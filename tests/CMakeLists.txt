cmake_minimum_required(VERSION 3.14)
project(correlation_test C)
set(CMAKE_C_STANDARD 11)
set(HOST ON CACHE BOOL "" FORCE)
set(DISABLEFLOAT16 ON CACHE BOOL "" FORCE)
set(CMSISDSP_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(../external/CMSIS-DSP dsp)
get_target_property(dsp_sources CMSISDSP SOURCES)
list(FILTER dsp_sources INCLUDE REGEX "(TransformFunctions/.*(f32)\\.c|TransformFunctions/arm_bitreversal[2]?\\.c|CommonTables/.*\\.c|FilteringFunctions/arm_fir(_init)?_f32\\.c|SupportFunctions/arm_(copy|fill)_.*\\.c)$")
set_property(TARGET CMSISDSP PROPERTY SOURCES "${dsp_sources}")
add_library(correlation STATIC ../src/correlation.c)
target_include_directories(correlation PUBLIC ../include)
target_link_libraries(correlation PUBLIC CMSISDSP)
add_executable(correlation_test correlation_test.c ../benchmarks/signals.c)
target_include_directories(correlation_test PRIVATE ../benchmarks)
target_link_libraries(correlation_test PRIVATE correlation)
if(UNIX AND NOT APPLE)
  target_link_libraries(correlation_test PRIVATE m)
endif()
enable_testing()
add_test(NAME correlation COMMAND correlation_test)
add_executable(first_echo_test first_echo_test.c ../benchmarks/signals.c)
target_include_directories(first_echo_test PRIVATE ../benchmarks)
target_link_libraries(first_echo_test PRIVATE correlation)
if(UNIX AND NOT APPLE)
  target_link_libraries(first_echo_test PRIVATE m)
endif()
add_test(NAME first_echo COMMAND first_echo_test)

cmake_minimum_required(VERSION 3.14)
project(correlation_test C)
set(CMAKE_C_STANDARD 11)
set(HOST ON CACHE BOOL "" FORCE)
set(DISABLEFLOAT16 ON CACHE BOOL "" FORCE)
set(CMSISDSP_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(../external/CMSIS-DSP dsp)
add_library(correlation STATIC ../src/correlation.c)
target_include_directories(correlation PUBLIC ../include)
target_link_libraries(correlation PUBLIC CMSISDSP)
add_executable(correlation_test correlation_test.c)
target_include_directories(correlation_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(correlation_test PRIVATE correlation)
if(UNIX AND NOT APPLE)
  target_link_libraries(correlation_test PRIVATE m)
endif()
enable_testing()
add_test(NAME correlation COMMAND correlation_test)
